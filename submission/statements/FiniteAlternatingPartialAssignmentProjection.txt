              ∀ i ∈ a, ∃ c ∈ ψ, ∃ l ∈ c, l.1 = i
          let res := fun (a : List ℕ) (σ : Assignment) =>

            ψ.filterMap fun c =>
              if c.any (fun l => ρ l.1 == some l.2) then none
              else some (c.filter fun l => (ρ l.1).isNone)
          if q.1 then ∀ a : List ℕ, ∀ σ : Assignment, admissible a → next (res a σ)
