open PEqualNP.Library PEqualNP.Search in
∀ p : Polynomial ℕ,
    ∃ L : Language Bool, L ∈ P ∧ (∀ x, x ∈ L → x ∈ SAT) ∧
      ∀ φ : CNF,
        2 ^ (φ.flatMap (fun c => if 2 < c.length then c.map Prod.fst else [])).toFinset.card ≤
          p.eval (encodeCNF φ).length →
        (encodeCNF φ ∈ L ↔ Satisfiable φ)
