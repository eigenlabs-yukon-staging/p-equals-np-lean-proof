open PEqualNP.Library PEqualNP.Search in
∀ (p : Polynomial ℕ),
    ∃ L : Language Bool, L ∈ P ∧ (∀ x, x ∈ L → x ∈ SAT) ∧
      ∀ φ : CNF,
        (∃ a : Finset ℕ,
          (∀ c ∈ φ, (c.filter (fun l => l.2 && decide (l.1 ∉ a))).length ≤ 1) ∧
          4 ^ a.card ≤ p.eval (encodeCNF φ).length) →
        (encodeCNF φ ∈ L ↔ Satisfiable φ)
